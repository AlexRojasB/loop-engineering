using System;
using System.Collections.Generic;
using System.Linq;

public enum OrderStatus
{
    Pending,
    Paid,
    Cancelled,
    Refunded
}

public class Order
{
    public Guid Id { get; }
    public decimal Total { get; }
    public string CustomerEmail { get; }
    public OrderStatus Status { get; set; }

    public Order(decimal total, string customerEmail)
    {
        if (string.IsNullOrWhiteSpace(customerEmail))
            throw new ArgumentException("Customer email cannot be null or whitespace.");

        if (total <= 0)
            throw new ArgumentException("Total must be greater than zero.");

        Id = Guid.NewGuid();
        Total = total;
        CustomerEmail = customerEmail;
        Status = OrderStatus.Pending;
    }
}

public class OrderService
{
    private readonly Dictionary<Guid, Order> _orders = new Dictionary<Guid, Order>();

    public bool CreateOrder(decimal total, string customerEmail)
    {
        try
        {
            var order = new Order(total, customerEmail);
            _orders[order.Id] = order;
            return true;
        }
        catch (ArgumentException)
        {
            return false;
        }
    }

    public bool PayOrder(Guid orderId)
    {
        if (!_orders.TryGetValue(orderId, out var order))
            return false;

        if (order.Status == OrderStatus.Pending)
        {
            order.Status = OrderStatus.Paid;
            return true;
        }

        return false;
    }

    public bool CancelOrder(Guid orderId)
    {
        if (!_orders.TryGetValue(orderId, out var order))
            return false;

        if (order.Status == OrderStatus.Pending)
        {
            order.Status = OrderStatus.Cancelled;
            return true;
        }

        return false;
    }

    public bool RefundOrder(Guid orderId)
    {
        if (!_orders.TryGetValue(orderId, out var order))
            return false;

        if (order.Status != OrderStatus.Paid)
            return false;

        order.Status = OrderStatus.Refunded;
        return true;
    }

    public List<Order> GetOrdersByCustomer(string customerEmail)
    {
        return _orders.Values.Where(o => string.Equals(o.CustomerEmail, customerEmail, StringComparison.OrdinalIgnoreCase)).ToList();
    }

    public static void Main(string[] args)
    {
        // This is a placeholder for the Main method. It should not be deleted.
        // It is required for the project to build successfully.
    }
}

        }

            return true;

using System;
using System.Collections.Generic;
using System.Linq;

public enum OrderStatus
{
    Pending,
    Paid,
    Cancelled,
    Refunded
}

public class Order
{
    public Guid Id { get; }
    public decimal Total { get; }
    public string CustomerEmail { get; }
    public OrderStatus Status { get; set; }

    public Order(decimal total, string customerEmail)
    {
        if (string.IsNullOrWhiteSpace(customerEmail))
            throw new ArgumentException("Customer email cannot be null or whitespace.");

        if (total <= 0)
            throw new ArgumentException("Total must be greater than zero.");

        Id = Guid.NewGuid();
        Total = total;
        CustomerEmail = customerEmail;
        Status = OrderStatus.Pending;
    }
}

public class OrderService
{
    private readonly Dictionary<Guid, Order> _orders = new Dictionary<Guid, Order>();

    public bool CreateOrder(decimal total, string customerEmail)
    {
        try
        {
            var order = new Order(total, customerEmail);
            _orders[order.Id] = order;
            return true;
        }
        catch (ArgumentException)
        {
            return false;
        }
    }

    public bool PayOrder(Guid orderId)
    {
        if (!_orders.TryGetValue(orderId, out var order))
            return false;

        if (order.Status == OrderStatus.Pending)
        {
            order.Status = OrderStatus.Paid;
            return true;
        }

        return false;
    }

    public bool CancelOrder(Guid orderId)
    {
        if (!_orders.TryGetValue(orderId, out var order))
            return false;

        if (order.Status == OrderStatus.Pending)
        {
            order.Status = OrderStatus.Cancelled;
            return true;
        }

        return false;
    }

    public bool RefundOrder(Guid orderId)
    {
        if (!_orders.TryGetValue(orderId, out var order))
            return false;

        if (order.Status != OrderStatus.Paid)
            return false;

        order.Status = OrderStatus.Refunded;
        return true;
    }

    public List<Order> GetOrdersByCustomer(string customerEmail)
    {
        return _orders.Values.Where(o => string.Equals(o.CustomerEmail, customerEmail, StringComparison.OrdinalIgnoreCase)).ToList();
    }

    }

using System;
using System.Collections.Generic;
using System.Linq;

public enum OrderStatus
{
    Pending,
    Paid,
    Cancelled,
    Refunded
}

public class Order
{
    public Guid Id { get; }
    public decimal Total { get; }
    public string CustomerEmail { get; }
    public OrderStatus Status { get; set; }

    public Order(decimal total, string customerEmail)
    {
        if (string.IsNullOrWhiteSpace(customerEmail))
            throw new ArgumentException("Customer email cannot be null or whitespace.");

        if (total <= 0)
            throw new ArgumentException("Total must be greater than zero.");

        Id = Guid.NewGuid();
        Total = total;
        CustomerEmail = customerEmail;
        Status = OrderStatus.Pending;
    }
}

public class OrderService
{
    private readonly Dictionary<Guid, Order> _orders = new Dictionary<Guid, Order>();

    public bool CreateOrder(decimal total, string customerEmail)
    {
        try
        {
            var order = new Order(total, customerEmail);
            _orders[order.Id] = order;
            return true;
        }
        catch (ArgumentException)
        {
            return false;
        }
    }

    public bool PayOrder(Guid orderId)
    {
        if (!_orders.TryGetValue(orderId, out var order))
            return false;

        if (order.Status == OrderStatus.Pending)
        {
            order.Status = OrderStatus.Paid;
            return true;
        }

        return false;
    }

    public bool CancelOrder(Guid orderId)
    {
        if (!_orders.TryGetValue(orderId, out var order))
            return false;

        if (order.Status == OrderStatus.Pending)
        {
            order.Status = OrderStatus.Cancelled;
            return true;
        }

        return false;
    }

    public bool RefundOrder(Guid orderId)
    {
        if (!_orders.TryGetValue(orderId, out var order))
            return false;

        if (order.Status == OrderStatus.Pending)
            return false;

        if (order.Status == OrderStatus.Cancelled)
            return false;

        if (order.Status == OrderStatus.Refunded)
            return false;

        if (order.Status == OrderStatus.Paid)
        {
            order.Status = OrderStatus.Refunded;
            return true;
        }

        return false;
    }

    public List<Order> GetOrdersByCustomer(string customerEmail)
    {
        return _orders.Values.Where(o => string.Equals(o.CustomerEmail, customerEmail, StringComparison.OrdinalIgnoreCase)).ToList();
    }
}

    public OrderStatus Status { get; set; }

    public Order(decimal total, string customerEmail)
    {
        if (string.IsNullOrWhiteSpace(customerEmail))
            throw new ArgumentException("Customer email cannot be null or whitespace.");

        if (total <= 0)
            throw new ArgumentException("Total must be greater than zero.");

        Id = Guid.NewGuid();
        Total = total;
        CustomerEmail = customerEmail;
        Status = OrderStatus.Pending;
    }
}

public class OrderService
{
    private readonly Dictionary<Guid, Order> _orders = new Dictionary<Guid, Order>();

    public bool CreateOrder(decimal total, string customerEmail)
    {
        try
        {
            var order = new Order(total, customerEmail);
            _orders[order.Id] = order;
            return true;
        }
        catch (ArgumentException)
        {
            return false;
        }
    }

    public bool PayOrder(Guid orderId)
    {
        if (!_orders.TryGetValue(orderId, out var order))
            return false;

        if (order.Status == OrderStatus.Pending)
        {
            order.Status = OrderStatus.Paid;
            return true;
        }

        return false;
    }

    public bool CancelOrder(Guid orderId)
    {
        if (!_orders.TryGetValue(orderId, out var order))
            return false;

        if (order.Status == OrderStatus.Pending)
        {
            order.Status = OrderStatus.Cancelled;
            return true;
        }

        return false;
    }

    public bool RefundOrder(Guid orderId)
    {
        if (!_orders.TryGetValue(orderId, out var order))
            return false;

        if (order.Status == OrderStatus.Pending)
            return false;

        if (order.Status == OrderStatus.Cancelled)
            return false;

        if (order.Status == OrderStatus.Refunded)
            return false;

        if (order.Status == OrderStatus.Paid)
        {
            order.Status = OrderStatus.Refunded;
            return true;
        }

        return false;
    }

    public List<Order> GetOrdersByCustomer(string customerEmail)
    {
        return _orders.Values.Where(o => string.Equals(o.CustomerEmail, customerEmail, StringComparison.OrdinalIgnoreCase)).ToList();
    }
}
