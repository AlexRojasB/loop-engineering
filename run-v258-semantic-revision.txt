using System;
using System.Linq;
using Xunit;

public class OrderServiceTests
{
    [Fact]
    public void CreateOrder_RejectsNullOrWhitespaceEmail()
    {
        var service = new OrderService();

        Assert.False(service.CreateOrder(100m, null!));
        Assert.False(service.CreateOrder(100m, ""));
        Assert.False(service.CreateOrder(100m, "   "));
    }

    [Fact]
    public void CreateOrder_RejectsZeroOrNegativeTotal()
    {
        var service = new OrderService();

        Assert.False(service.CreateOrder(0m, "test@example.com"));
        Assert.False(service.CreateOrder(-1m, "test@example.com"));
    }

    [Fact]
    public void CreateOrder_AssignsUniqueIds_AndPendingStatus()
    {
        var service = new OrderService();

        Assert.True(service.CreateOrder(100m, "a@example.com"));
        Assert.True(service.CreateOrder(200m, "b@example.com"));

        var first = service.GetOrdersByCustomer("a@example.com").Single();
        var second = service.GetOrdersByCustomer("b@example.com").Single();

        Assert.NotEqual(Guid.Empty, first.Id);
        Assert.NotEqual(Guid.Empty, second.Id);
        Assert.NotEqual(first.Id, second.Id);

        Assert.Equal(OrderStatus.Pending, first.Status);
        Assert.Equal(OrderStatus.Pending, second.Status);
    }

    [Fact]
    public void PayOrder_ReturnsFalse_WhenOrderDoesNotExist()
    {
        var service = new OrderService();

        Assert.False(service.PayOrder(Guid.NewGuid()));
    }

    [Fact]
    public void PayOrder_TransitionsPendingToPaid()
    {
        var service = new OrderService();

        service.CreateOrder(100m, "pay@example.com");
        var order = service.GetOrdersByCustomer("pay@example.com").Single();

        Assert.True(service.PayOrder(order.Id));
        Assert.Equal(OrderStatus.Paid, order.Status);
    }

    [Fact]
    public void PayOrder_DoesNotAllowAlreadyPaidOrder()
    {
        var service = new OrderService();

        service.CreateOrder(100m, "paid@example.com");
        var order = service.GetOrdersByCustomer("paid@example.com").Single();

        Assert.True(service.PayOrder(order.Id));
        Assert.False(service.PayOrder(order.Id));
    }

    [Fact]
    public void PayOrder_DoesNotAllowCancelledOrder()
    {
        var service = new OrderService();

        service.CreateOrder(100m, "cancelled@example.com");
        var order = service.GetOrdersByCustomer("cancelled@example.com").Single();

        Assert.True(service.CancelOrder(order.Id));
        Assert.False(service.PayOrder(order.Id));
    }

    [Fact]
    public void CancelOrder_ReturnsFalse_WhenOrderDoesNotExist()
    {
        var service = new OrderService();

        Assert.False(service.CancelOrder(Guid.NewGuid()));
    }

    [Fact]
    public void CancelOrder_TransitionsPendingToCancelled()
    {
        var service = new OrderService();

        service.CreateOrder(100m, "cancel@example.com");
        var order = service.GetOrdersByCustomer("cancel@example.com").Single();

        Assert.True(service.CancelOrder(order.Id));
        Assert.Equal(OrderStatus.Cancelled, order.Status);
    }

    [Fact]
    public void CancelOrder_DoesNotAllowPaidOrder()
    {
        var service = new OrderService();

        service.CreateOrder(100m, "cannotcancel@example.com");
        var order = service.GetOrdersByCustomer("cannotcancel@example.com").Single();

        Assert.True(service.PayOrder(order.Id));
        Assert.False(service.CancelOrder(order.Id));
    }

    [Fact]
    public void GetOrdersByCustomer_IsCaseInsensitive()
    {
        var service = new OrderService();

        service.CreateOrder(100m, "Test@Example.com");

        var result = service.GetOrdersByCustomer("test@example.com");

        Assert.Single(result);
    }

    [Fact]
    public void GetOrdersByCustomer_DoesNotExposeInternalCollection()
    {
        var service = new OrderService();

        service.CreateOrder(100m, "copy@example.com");

        var result = service.GetOrdersByCustomer("copy@example.com");

        var list = result.ToList();
        list.Clear();

        var secondRead = service.GetOrdersByCustomer("copy@example.com");

        Assert.Single(secondRead);
    }

    [Fact]
    public void RefundOrder_ReturnsFalse_WhenOrderDoesNotExist()
    {
        var service = new OrderService();

        Assert.False(service.RefundOrder(Guid.NewGuid()));
    }

    [Fact]
    public void RefundOrder_ReturnsFalse_WhenOrderIsPending()
    {
        var service = new OrderService();

        service.CreateOrder(100m, "pending@example.com");
        var order = service.GetOrdersByCustomer("pending@example.com").Single();

        Assert.False(service.RefundOrder(order.Id));
    }

    [Fact]
    public void RefundOrder_ReturnsFalse_WhenOrderIsCancelled()
    {
        var service = new OrderService();

        service.CreateOrder(100m, "cancelled@example.com");
        var order = service.GetOrdersByCustomer("cancelled@example.com").Single();
        service.CancelOrder(order.Id);

        Assert.False(service.RefundOrder(order.Id));
    }

    [Fact]
    public void RefundOrder_ReturnsFalse_WhenOrderIsAlreadyRefunded()
    {
        var service = new OrderService();

        service.CreateOrder(100m, "alreadyrefunded@example.com");
        var order = service.GetOrdersByCustomer("alreadyrefunded@example.com").Single();
        service.PayOrder(order.Id);
        service.RefundOrder(order.Id);

        Assert.False(service.RefundOrder(order.Id));
    }

    [Fact]
    public void RefundOrder_ReturnsTrue_WhenOrderIsPaid()
    {
        var service = new OrderService();

        service.CreateOrder(100m, "paid@example.com");
        var order = service.GetOrdersByCustomer("paid@example.com").Single();
        service.PayOrder(order.Id);

        Assert.True(service.RefundOrder(order.Id));
    }

    [Fact]
    public void RefundOrder_ChangesOrderStatusToRefunded()
    {
        var service = new OrderService();

        service.CreateOrder(100m, "status@example.com");
        var order = service.GetOrdersByCustomer("status@example.com").Single();
        service.PayOrder(order.Id);

        service.RefundOrder(order.Id);

        Assert.Equal(OrderStatus.Refunded, order.Status);
    }

    [Fact]
    public void RefundOrder_DoesNotChangeStatusOnSecondCall()
    {
        var service = new OrderService();

        service.CreateOrder(100m, "secondcall@example.com");
        var order = service.GetOrdersByCustomer("secondcall@example.com").Single();
        service.PayOrder(order.Id);
        service.RefundOrder(order.Id);

        Assert.False(service.RefundOrder(order.Id));
        Assert.Equal(OrderStatus.Refunded, order.Status);
    }

    [Fact]
    public void RefundOrder_FailedRefundDoesNotModifyStatus()
    {
        var service = new OrderService();

        service.CreateOrder(100m, "failedrefund@example.com");
        var order = service.GetOrdersByCustomer("failedrefund@example.com").Single();
        service.PayOrder(order.Id);
        service.RefundOrder(order.Id);

        // Assuming the refund process fails for some reason
        service.RefundOrder(order.Id); // This should still return false

        Assert.Equal(OrderStatus.Refunded, order.Status);
    }
